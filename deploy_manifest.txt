# deploy_manifest.txt - the one list of what deploy.py ships from this project to a runtime install
# (usually <Morrowind>\conflict-tools\). Credit: dej (dejunai).
#
# Only files git tracks are candidates. Rules, one per line, paths relative to the project with '/':
#   include <glob>                  ship the matching files at the same path in the target
#   include <file> => <target path> ship one file under another path (a relative link in a moved doc would break)
#   exclude <glob>                  never ship these (an exclude always wins over an include)
# Globs: '*' stays inside one folder, '**' crosses folders ("lua tools/**" is everything below "lua tools").
# deploy.py also refuses on its own, whatever this file says: tests, .git and CI files, deploy.py and its
# deploy.ini/manifest, doc_scan.py, sync.py, tes3diff.py, the dummy stand-in folder (generated at run time), generated
# records, audits, backups, docs/dev and personal notes (see NEVER in deploy.py).

# The tools: the Python modules and launchers at the project root
include *.py
exclude doc_scan.py                     # dev: verifies documented switches against the code
exclude sync.py                         # dev: repository -> MWManager; never writes to an install
exclude tes3diff.py                     # dev: record-level plugin diff through the optional tes3conv
exclude dummy assets/**                 # stand-ins are made in code or copied out of Morrowind.bsa at run time:
exclude **/dummy/**                     # never shipped (they live in <tools>\generated\dummy\)
include *.bat
exclude Play Morrowind.bat              # the game-folder launcher: it lives beside Morrowind.exe, not in conflict-tools

# What the tools deploy and run: the suite's MWSE Lua (laid out as in Data Files) and the bundled programs
include lua tools/**
exclude lua tools/**/MWSE/core/**       # MWSE's own core files are never ours to ship (luafix patches them in place)

include tools/**

# Docs keep the paths they have in the repository, so a relative link that works here works in the install.
include docs/**
exclude docs/dev/**                     # design notes, TODO and the Lua repair log: development only
include README.md                       # stays at the root, as on GitHub
include LICENSE
